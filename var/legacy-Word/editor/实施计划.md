```python
#!/usr/bin/env python3
"""公文所见即所得编辑器 —— 本地后端

用法:
    python app.py [端口]
启动后浏览器打开 http://127.0.0.1:8765
依赖: python-docx（tools/docx2md.py、tools/md2docx.py 已需要）
"""
import hashlib
import json
import os
import re
import shutil
import subprocess
import sys
import time
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from urllib.parse import urlparse, parse_qs

BASE = os.path.dirname(os.path.abspath(__file__))
TOOLS = os.path.normpath(os.path.join(BASE, "..", "tools"))
WORK = os.path.join(BASE, "work")
SNAP_DIR = os.path.join(WORK, "快照")
CUR_MD = os.path.join(WORK, "当前.md")
CUR_DOCX = os.path.join(WORK, "输出.docx")
KEEP_SNAPS = 20

os.makedirs(WORK, exist_ok=True)
os.makedirs(SNAP_DIR, exist_ok=True)


def run_converter(script, *args):
    """调用 tools/ 下的转换脚本，返回 (returncode, stdout, stderr)。"""
    p = subprocess.run(
        [sys.executable, os.path.join(TOOLS, script), *args],
        capture_output=True, text=True,
    )
    return p.returncode, p.stdout, p.stderr


def atomic_write(path, text):
    """先写 .tmp 再 os.replace，写一半断电也不会产生半截文件。"""
    tmp = path + ".tmp"
    with open(tmp, "w", encoding="utf-8") as f:
        f.write(text)
    os.replace(tmp, path)


def snapshot():
    """把当前.md 备份一份带时间戳的快照，最多保留 KEEP_SNAPS 份。"""
    if not os.path.exists(CUR_MD):
        return
    ts = time.strftime("%Y%m%d_%H%M%S")
    shutil.copy2(CUR_MD, os.path.join(SNAP_DIR, f"快照_{ts}.md"))
    snaps = sorted(f for f in os.listdir(SNAP_DIR) if f.startswith("快照_"))
    for old in snaps[:-KEEP_SNAPS]:
        os.remove(os.path.join(SNAP_DIR, old))


def read_md():
    if not os.path.exists(CUR_MD):
        return ""
    with open(CUR_MD, "r", encoding="utf-8") as f:
        return f.read()


def md_hash():
    if not os.path.exists(CUR_MD):
        return ""
    with open(CUR_MD, "rb") as f:
        return hashlib.md5(f.read()).hexdigest()


def parse_multipart(raw, boundary):
    """极简 multipart/form-data 解析，只取第一个带 filename 的字段。"""
    if not boundary:
        return None, None
    delimiter = b"--" + boundary
    for part in raw.split(delimiter):
        if b"filename=" not in part:
            continue
        head, _, body = part.partition(b"\r\n\r\n")
        m = re.search(rb'filename="([^"]*)"', head)
        if not m:
            continue
        data = body.rstrip(b"\r\n")
        if data.endswith(b"\r\n"):
            data = data[:-2]
        return m.group(1).decode("utf-8", errors="replace"), data
    return None, None


CONTENT_TYPES = {
    ".html": "text/html; charset=utf-8",
    ".css": "text/css; charset=utf-8",
    ".js": "application/javascript; charset=utf-8",
    ".svg": "image/svg+xml",
    ".png": "image/png",
    ".gif": "image/gif",
    ".woff2": "font/woff2",
    ".woff": "font/woff",
}


class Handler(BaseHTTPRequestHandler):
    def log_message(self, fmt, *args):
        pass

    def _json(self, obj, code=200):
        body = json.dumps(obj, ensure_ascii=False).encode("utf-8")
        self.send_response(code)
        self.send_header("Content-Type", "application/json; charset=utf-8")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def _file(self, path, content_type):
        path = os.path.normpath(path)
        if not (path == BASE or path.startswith(BASE + os.sep)) or not os.path.isfile(path):
            self._json({"error": "not found"}, 404)
            return
        with open(path, "rb") as f:
            body = f.read()
        self.send_response(200)
        self.send_header("Content-Type", content_type)
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def _static(self, url_path):
        if url_path in ("/", "/editor.html"):
            self._file(os.path.join(BASE, "editor.html"), CONTENT_TYPES[".html"])
        elif url_path == "/gongwen.css":
            self._file(os.path.join(BASE, "gongwen.css"), CONTENT_TYPES[".css"])
        elif url_path.startswith("/vditor/"):
            rel = url_path[len("/vditor/"):]
            ext = os.path.splitext(rel)[1].lower()
            self._file(os.path.join(BASE, "vendor", "vditor", rel),
                       CONTENT_TYPES.get(ext, "application/octet-stream"))
        else:
            self._json({"error": "not found"}, 404)

    def do_GET(self):
        url = urlparse(self.path)
        if url.path.startswith("/api/"):
            self._api_get(url)
        else:
            self._static(url.path)

    def do_POST(self):
        url = urlparse(self.path)
        if url.path.startswith("/api/"):
            self._api_post(url)
        else:
            self._json({"error": "not found"}, 404)

    def _api_get(self, url):
        if url.path == "/api/content":
            self._json({"md": read_md(), "hash": md_hash()})
        elif url.path == "/api/poll":
            self._json({"hash": md_hash()})
        elif url.path == "/api/export":
            qs = parse_qs(url.query)
            fmt = qs.get("fmt", ["docx"])[0]
            if fmt == "docx":
                if not os.path.exists(CUR_MD):
                    self._json({"error": "还没有内容可导出"}, 400)
                    return
                rc, _, err = run_converter("md2docx.py", CUR_MD, CUR_DOCX)
                if rc != 0:
                    self._json({"error": f"md2docx 转换失败：{err.strip()}（请确认 Word 未占用 work/输出.docx）"}, 500)
                    return
                with open(CUR_DOCX, "rb") as f:
                    body = f.read()
                self.send_response(200)
                self.send_header("Content-Type",
                                 "application/vnd.openxmlformats-officedocument.wordprocessingml.document")
                self.send_header("Content-Disposition", 'attachment; filename="输出.docx"')
                self.send_header("Content-Length", str(len(body)))
                self.end_headers()
                self.wfile.write(body)
            else:
                body = read_md().encode("utf-8")
                self.send_response(200)
                self.send_header("Content-Type", "text/markdown; charset=utf-8")
                self.send_header("Content-Disposition", 'attachment; filename="输出.md"')
                self.send_header("Content-Length", str(len(body)))
                self.end_headers()
                self.wfile.write(body)
        else:
            self._json({"error": "not found"}, 404)

    def _api_post(self, url):
        if url.path == "/api/save":
            length = int(self.headers.get("Content-Length", 0))
            raw = self.rfile.read(length)
            try:
                md = json.loads(raw.decode("utf-8")).get("md", "")
            except Exception:
                self._json({"error": "请求体不是合法 JSON"}, 400)
                return
            snapshot()
            atomic_write(CUR_MD, md)
            self._json({"ok": True, "hash": md_hash()})
        elif url.path == "/api/open":
            length = int(self.headers.get("Content-Length", 0))
            raw = self.rfile.read(length)
            boundary = self.headers.get("Content-Type", "").split("boundary=", 1)[-1].encode()
            filename, data = parse_multipart(raw, boundary)
            if not filename:
                self._json({"error": "没有收到文件"}, 400)
                return
            ext = os.path.splitext(filename)[1].lower()
            if ext not in (".docx", ".md"):
                self._json({"error": f"仅支持 .docx 或 .md，收到的是 {ext or '未知类型'}"}, 400)
                return
            snapshot()  # 打开新文档前备份当前内容
            src = os.path.join(WORK, "上传" + ext)
            with open(src, "wb") as f:
                f.write(data)
            if ext == ".docx":
                tmp_md = CUR_MD + ".conv.tmp"
                rc, _, err = run_converter("docx2md.py", src, tmp_md)
                if rc != 0:
                    if os.path.exists(tmp_md):
                        os.remove(tmp_md)
                    self._json({"error": f"docx2md 转换失败：{err.strip()}"}, 500)
                    return
                os.replace(tmp_md, CUR_MD)
            else:
                atomic_write(CUR_MD, data.decode("utf-8", errors="replace"))
            self._json({"ok": True, "filename": filename, "md": read_md(), "hash": md_hash()})
        else:
            self._json({"error": "not found"}, 404)


def main():
    port = int(sys.argv[1]) if len(sys.argv) > 1 else 8765
    server = ThreadingHTTPServer(("127.0.0.1", port), Handler)
    print(f"公文所见即所得编辑器已启动： http://127.0.0.1:{port}")
    print("浏览器打开上面的地址。按 Ctrl+C 停止。")
    server.serve_forever()


if __name__ == "__main__":
    main()
```

"""app.py 单元与集成测试（标准库 unittest，无第三方依赖）。"""
import http.client
import json
import os
import tempfile
import threading
import unittest

import app


class TestCore(unittest.TestCase):
    def test_atomic_write_creates_and_replaces(self):
        with tempfile.TemporaryDirectory() as d:
            p = os.path.join(d, "x.md")
            app.atomic_write(p, "第一版")
            self.assertEqual(open(p, encoding="utf-8").read(), "第一版")
            app.atomic_write(p, "第二版")
            self.assertEqual(open(p, encoding="utf-8").read(), "第二版")
            self.assertFalse(os.path.exists(p + ".tmp"))

    def test_snapshot_keeps_cap(self):
        with tempfile.TemporaryDirectory() as d:
            cur = os.path.join(d, "cur.md")
            snap = os.path.join(d, "snap")
            os.makedirs(snap)
            open(cur, "w", encoding="utf-8").write("内容")
            old = (app.CUR_MD, app.SNAP_DIR)
            app.CUR_MD, app.SNAP_DIR = cur, snap
            try:
                for _ in range(app.KEEP_SNAPS + 5):
                    app.snapshot()
                count = len([f for f in os.listdir(snap) if f.startswith("快照_")])
                self.assertEqual(count, app.KEEP_SNAPS)
            finally:
                app.CUR_MD, app.SNAP_DIR = old

    def test_parse_multipart(self):
        boundary = "abc123"
        raw = (
            b"--abc123\r\n"
            b'Content-Disposition: form-data; name="file"; filename="测试.docx"\r\n'
            b"Content-Type: application/octet-stream\r\n\r\n"
            b"BINARYDATA\r\n"
            b"--abc123--\r\n"
        )
        name, data = app.parse_multipart(raw, boundary.encode())
        self.assertEqual(name, "测试.docx")
        self.assertEqual(data, b"BINARYDATA")


class TestRoundtrip(unittest.TestCase):
    def test_md_docx_md_roundtrip(self):
        md = (
            "# 2026年上半年个人工作总结\n\n"
            "## 一、坚持政治引领\n\n"
            "### （一）加强理论学习\n\n"
            "**一是统筹推进。**始终坚持第一议题学习。\n\n"
            "| 序号 | 项目 |\n|------|------|\n| 1 | 大冬会 |\n\n"
            '<div align="right">二〇二六年八月</div>\n'
        )
        with tempfile.TemporaryDirectory() as d:
            m1 = os.path.join(d, "a.md")
            d1 = os.path.join(d, "a.docx")
            m2 = os.path.join(d, "b.md")
            open(m1, "w", encoding="utf-8").write(md)
            rc, _, err = app.run_converter("md2docx.py", m1, d1)
            self.assertEqual(rc, 0, err)
            rc, _, err = app.run_converter("docx2md.py", d1, m2)
            self.assertEqual(rc, 0, err)
            out = open(m2, encoding="utf-8").read()
            for key in ["2026年上半年个人工作总结", "一、坚持政治引领", "（一）加强理论学习",
                        "统筹推进", "大冬会", "二〇二六年八月", "序号", "1"]:
                self.assertIn(key, out)


class TestServer(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.workdir = tempfile.mkdtemp()
        cls.old = (app.WORK, app.SNAP_DIR, app.CUR_MD, app.CUR_DOCX)
        app.WORK = cls.workdir
        app.SNAP_DIR = os.path.join(cls.workdir, "快照")
        app.CUR_MD = os.path.join(cls.workdir, "当前.md")
        app.CUR_DOCX = os.path.join(cls.workdir, "输出.docx")
        os.makedirs(app.SNAP_DIR, exist_ok=True)
        cls.server = app.ThreadingHTTPServer(("127.0.0.1", 0), app.Handler)
        cls.port = cls.server.server_address[1]
        cls.thread = threading.Thread(target=cls.server.serve_forever, daemon=True)
        cls.thread.start()

    @classmethod
    def tearDownClass(cls):
        cls.server.shutdown()
        app.WORK, app.SNAP_DIR, app.CUR_MD, app.CUR_DOCX = cls.old

    def _request(self, method, path, body=None, headers=None):
        conn = http.client.HTTPConnection("127.0.0.1", self.port)
        conn.request(method, path, body=body, headers=headers or {})
        res = conn.getresponse()
        data = res.read()
        conn.close()
        return res.status, data

    def test_save_and_content(self):
        status, _ = self._request("POST", "/api/save",
                                  json.dumps({"md": "你好，公文"}).encode("utf-8"),
                                  {"Content-Type": "application/json"})
        self.assertEqual(status, 200)
        status, data = self._request("GET", "/api/content")
        self.assertEqual(status, 200)
        self.assertEqual(json.loads(data)["md"], "你好，公文")
        self.assertTrue(json.loads(data)["hash"])

    def test_export_docx(self):
        self._request("POST", "/api/save",
                      json.dumps({"md": "# 标题\n\n正文"}).encode("utf-8"),
                      {"Content-Type": "application/json"})
        status, data = self._request("GET", "/api/export?fmt=docx")
        self.assertEqual(status, 200)
        self.assertTrue(data.startswith(b"PK"))
        self.assertGreater(len(data), 100)


if __name__ == "__main__":
    unittest.main()

  <script>
    const $ = (id) => document.getElementById(id);
    const statusEl = $('status');
    const fileInput = $('fileInput');

    let vditor = null;
    let saveTimer = null;
    let dirty = false;
    let currentHash = '';

    function setStatus(msg, color) {
      statusEl.textContent = msg;
      statusEl.style.color = color || '#666';
    }

    async function api(url, opts) {
      const res = await fetch(url, opts);
      if (!res.ok) {
        let err = '请求失败';
        try { err = (await res.json()).error || err; } catch (e) {}
        throw new Error(err);
      }
      return res;
    }

    async function postSave() {
      const md = vditor.getValue();
      try {
        const res = await api('/api/save', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ md }),
        });
        const data = await res.json();
        currentHash = data.hash;
        dirty = false;
        setStatus('已自动保存到本地 md ✓', '#2e7d32');
      } catch (e) {
        setStatus('保存失败：' + e.message, '#c62828');
      }
    }

    function scheduleSave() {
      dirty = true;
      setStatus('编辑中…（自动保存）', '#ff8f00');
      clearTimeout(saveTimer);
      saveTimer = setTimeout(postSave, 1200);
    }

    async function loadMd(md, hash) {
      vditor.setValue(md);
      currentHash = hash;
      dirty = false;
    }

    async function openFile(file) {
      const form = new FormData();
      form.append('file', file);
      setStatus('后台转换中…', '#ff8f00');
      try {
        const res = await api('/api/open', { method: 'POST', body: form });
        const data = await res.json();
        await loadMd(data.md, data.hash);
        setStatus('已打开：' + data.filename + '（自动保存已开启）', '#2e7d32');

      }
